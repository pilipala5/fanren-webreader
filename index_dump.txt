    async function fetchJSON(url){ const r=await fetch(url); if(!r.ok) throw new Error('\u52a0\u8f7d\u5931\u8d25'); return await r.json(); }
    async function authMe(){ if(!API_AVAILABLE) return null; try{ const r=await fetch('/api/auth/me'); if(!r.ok) return null; return await r.json(); }catch{return null;} }
    async function authLogin(username,password){ if(!API_AVAILABLE) return { ok:false, error:'api disabled' }; const r=await fetch('/api/auth/login',{method:'POST',headers:{'Content-Type':'application/json'},body:JSON.stringify({username,password})}); return await r.json(); }
    async function authLogout(){ if(!API_AVAILABLE) return; try{ await fetch('/api/auth/logout',{method:'POST'}); }catch{} }

    async function loadBuiltins(){ for(const b of BOOKS){ try{ state.books[b.key]=await fetchJSON(b.manifest); }catch{} } }

    async function loadProgress(){ if(API_AVAILABLE){ try{ const r=await fetch('/api/progress'); if(r.ok){ const j=await r.json(); if(j.ok) state.progress=j.items||{}; }}catch{} } }

    function bookCard({ title, subtitle, tag, href, pos }){
      const d=document.createElement('a'); d.className='book-card link'; d.href=href; d.innerHTML=
        `<div class="cover">${escapeHtml(title)}</div>
         <div class="body">

    function escapeHtml(s){ return String(s||'').replace(/[&<>"']/g, c=>({"&":"&amp;","<":"&lt;",">":"&gt;","\"":"&quot;","'":"&#39;"}[c])); }

    }

    async function readFileAsText(file){ return new Promise((res,rej)=>{ const fr=new FileReader(); fr.onerror=()=>rej(fr.error||new Error('\u8bfb\u53d6\u5931\u8d25')); fr.onload=()=>res(String(fr.result||'')); fr.readAsText(file); }); }
    function splitTxt(text){
      const lines=String(text).replace(/\r\n?/g,'\n').split('\n');
      const heads=[]; const re=/^\\s*(?:\\u7b2c[\\u96f6\\u4e00\\u4e8c\\u4e09\\u56db\\u4e94\\u516d\\u4e03\\u516b\\u4e5d\\u5341\\u767e\\u5343\\u4e07\\u4e240-9]+[\\u7ae0\\u8282\\u56de\\u5377\\u90e8\\u96c6]\\s*)?\\u7b2c?[\\u96f6\\u4e00\\u4e8c\\u4e09\\u56db\\u4e94\\u516d\\u4e03\\u516b\\u4e5d\\u5341\\u767e\\u5343\\u4e07\\u4e240-9]+[\\u7ae0\\u56de][^\\n\\r]*$/;
      for(let i=0;i<lines.length;i++){ const s=lines[i].trim(); if(!s) continue; if(re.test(s)) heads.push([s,i]); }
      const out=[]; const push=(i,t,b)=>out.push({index:i,title:t,body:b.join('\n').trimStart()});
      if(!heads.length){ push(1,'\u5168\u6587',lines); return out; }
      const firstStart=heads[0][1]; if(lines.slice(0,firstStart).some(x=>x.trim())) push(0,'\u5f00\u7bc7',lines.slice(0,firstStart));
      for(let i=0;i<heads.length;i++){ const [t,s]=heads[i]; const e=i+1<heads.length? heads[i+1][1] : lines.length; const body=lines.slice(s,e); if(!body.length||body[0].trim()!=t.trim()) body.unshift('',t); push(i+1,t,body); }
      return out;
    }
    async function handleTxt(file){ const text=await readFileAsText(file); const chapters=splitTxt(text); const id='u:'+Date.now()+':'+Math.random().toString(36).slice(2); const title=file.name.replace(/\.[^.]+$/,''); const mani={id,title,type:'txt',chapters:chapters.map(c=>({index:c.index,title:c.title}))}; sessionStorage.setItem('upload:'+id+':manifest', JSON.stringify(mani)); for(const ch of chapters){ sessionStorage.setItem('upload:'+id+':ch:'+ch.index, ch.body); } closeUpload(); renderGrid(); }

    // 头部登录状态渲�?    function renderAuth(){ const u=el('userName'); const btnIn=el('btnLogin'); const btnOut=el('btnLogout'); if(state.userName){ u.textContent=state.userName; btnIn.style.display='none'; btnOut.style.display='inline-block'; } else { u.textContent='\u672a\u767b\u5f55'; btnIn.style.display='inline-block'; btnOut.style.display='none'; } }
