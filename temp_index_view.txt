<!doctype html>
<html lang="zh-CN">
<head>
  <meta charset="utf-8" />
  <meta name="viewport" content="width=device-width, initial-scale=1" />
  <title>灵阅书屋 · 首页</title>
  <link rel="stylesheet" href="assets/style.css" />
  <script defer src="https://cdn.jsdelivr.net/npm/fflate@0.8.0/umd/index.js"></script>
  <script>
    const BUILTINS = [
      { key: '凡人修仙传', manifest: '凡人修仙传/manifest.json' },
      { key: '凡人修仙传·仙界篇', manifest: '凡人修仙传·仙界篇/manifest.json' },
    ];

    let API_AVAILABLE = false;
    const state = { books: {}, serverBooks: [], progress: {}, userName: null };
    const favs = new Set();

    const $ = (id) => document.getElementById(id);
    const escapeHtml = (s) => String(s||'').replace(/[&<>"']/g, c=>({"&":"&amp;","<":"&lt;",">":"&gt;","\"":"&quot;","'":"&#39;"}[c]));

    async function pingAPI(){ const c=new AbortController(); setTimeout(()=>c.abort(),600); try{ const r=await fetch('/api/health',{signal:c.signal}); API_AVAILABLE=r.ok; }catch{ API_AVAILABLE=false; } }
    async function fetchJSON(url){ const r=await fetch(url); if(!r.ok) throw new Error('加载失败: '+url); return await r.json(); }

    async function authMe(){ if(!API_AVAILABLE) return null; try{ const r=await fetch('/api/auth/me'); if(!r.ok) return null; return await r.json(); }catch{return null;} }
    async function authLogin(username,password){ if(!API_AVAILABLE) return { ok:false, error:'api disabled' }; const r=await fetch('/api/auth/login',{method:'POST',headers:{'Content-Type':'application/json'},body:JSON.stringify({username,password})}); return await r.json(); }
    async function authLogout(){ if(!API_AVAILABLE) return; try{ await fetch('/api/auth/logout',{method:'POST'}); }catch{} }

    async function loadBuiltins(){ for(const b of BUILTINS){ try{ state.books[b.key]=await fetchJSON(b.manifest); }catch{ /* skip broken manifest */ } } }
    async function loadServerBooks(){ if(!API_AVAILABLE) return; try{ const r=await fetch('/api/books'); if(!r.ok) return; const j=await r.json(); if(j&&j.ok) state.serverBooks=j.items||[]; }catch{} }
    async function loadProgress(){ if(API_AVAILABLE){ try{ const r=await fetch('/api/progress'); if(r.ok){ const j=await r.json(); if(j.ok) state.progress=j.items||{}; }}catch{} } }

    async function favLoad(){ if(!API_AVAILABLE) return; try{ const r=await fetch('/api/favorites'); if(!r.ok) return; const j=await r.json(); if(j&&j.ok){ favs.clear(); for(const it of (j.items||[])) favs.add(String(it.book)); } }catch{} }
    async function favToggle(bookKey){ if(!API_AVAILABLE){ alert('未连接服务器'); return; } if(!state.userName){ if(!await ensureLogin()) return; }
      const action=favs.has(bookKey)? 'remove' : 'add';
      try{ const r=await fetch('/api/favorites',{method:'POST',headers:{'Content-Type':'application/json'},body:JSON.stringify({action, book:bookKey})}); const j=await r.json(); if(r.ok&&j.ok){ if(action==='add') favs.add(bookKey); else favs.delete(bookKey); renderGrid(); } }catch{}
    }

    function bookCard({ key, title, subtitle, tag, href, pos }){
      const wrap=document.createElement('div'); wrap.className='book-card';
      const a=document.createElement('a'); a.className='link'; a.href=href; a.style.textDecoration='none';
      a.innerHTML = `
        <div class="cover">${escapeHtml(title)}</div>
        <div class="body">
          <div class="title">${escapeHtml(title)}</div>
          <div class="meta">${escapeHtml(subtitle||'')}</div>
          <div class="chips"><span class="chip">${escapeHtml(tag||'')}</span>${(Number.isFinite(pos)&&pos>0)? `<span class="chip">已读至 第 ${pos} 章</span>`:''}</div>
        </div>`;
      wrap.appendChild(a);
      const fav=document.createElement('button'); fav.type='button'; fav.className='fav-btn'+(favs.has(String(key))?' active':''); fav.innerHTML='<span>★</span>';
      fav.addEventListener('click', async (e)=>{ e.preventDefault(); e.stopPropagation(); await favToggle(String(key)); });
      wrap.appendChild(fav);
      return wrap;
    }

    function renderAuth(){ const u=$('userName'); const btnIn=$('btnLogin'); const btnOut=$('btnLogout'); if(state.userName){ u.textContent=state.userName; btnIn.style.display='none'; btnOut.style.display='inline-block'; } else { u.textContent='未登录'; btnIn.style.display='inline-block'; btnOut.style.display='none'; } }
    async function ensureLogin(){ if(!API_AVAILABLE) return false; if(state.userName) return true; const u=prompt('用户名'); if(!u) return false; const p=prompt('密码（≥6位）'); if(!p) return false; const r=await authLogin(u,p); if(r&&r.ok){ state.userName=u; renderAuth(); return true; } alert(r?.error||'登录失败'); return false; }

    function renderGrid(){
      const grid = $('bookGrid'); if(!grid) return; grid.innerHTML='';
      // 内置
      for(const b of BUILTINS){ const m=state.books[b.key]; if(!m) continue; const total=(m.chapters||[]).length; const k=b.key; const idx=Number(state.progress[k]||localStorage.getItem('novel:p:'+k)||0)||0; grid.appendChild(bookCard({ key:k, title:b.key, subtitle:`章节数 ${total}`, tag:'内置', href:'viewer.html?book='+encodeURIComponent(k), pos: idx })); }
      // 服务器
      for(const sb of state.serverBooks){ const k='srv:'+sb.id; const total=sb.chapters||0; const idx=Number(state.progress[k]||localStorage.getItem('novel:p:'+k)||0)||0; grid.appendChild(bookCard({ key:k, title:sb.title, subtitle:`章节数 ${total} · 上传者：${sb.uploader}`, tag:'服务器', href:'viewer.html?srv='+encodeURIComponent(sb.id), pos: idx })); }
    }

    async function init(){
      await pingAPI();
      const me = await authMe(); if(me&&me.ok&&me.loggedIn){ state.userName=me.username; }
      renderAuth();
      await loadBuiltins();
      await loadServerBooks();
      await loadProgress();
      if(API_AVAILABLE) await favLoad();
      renderGrid();
      $('btnLogin').addEventListener('click', async ()=>{ await ensureLogin(); if(API_AVAILABLE) await favLoad(); renderGrid(); });
      $('btnLogout').addEventListener('click', async ()=>{ await authLogout(); state.userName=null; renderAuth(); });
    }
    document.addEventListener('DOMContentLoaded', init);
  </script>
</head>

      <span class="small">账户：</span><strong id="userName" class="small">未登录</strong>
      <button class="btn" id="btnLogin">登录/注册</button>
      <button class="btn" id="btnLogout" style="display:none">退出</button>
    </div>
  </div>
